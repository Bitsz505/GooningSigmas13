#!/bin/bash

# --- Configuration ---
LOG_FILE="/var/log/system_updater.log"
DATE_TIME=$(date '+%Y-%m-%d %H:%M:%S')

# --- Functions ---

# Function to check for root privileges
check_root() {
    if [ "$EUID" -ne 0 ]; then
        echo "Error: This script must be run as root (use sudo)." | tee -a "$LOG_FILE"
        exit 1
    fi
}

# Function to execute a command and log its output
run_command() {
    local cmd="$1"
    echo "[$DATE_TIME] Executing: $cmd" | tee -a "$LOG_FILE"
    if ! eval "$cmd" >> "$LOG_FILE" 2>&1; then
        echo "[$DATE_TIME] ERROR: Command failed." | tee -a "$LOG_FILE"
        return 1
    fi
    return 0
}

# --- Main Execution ---

check_root

echo "--- Starting System Update Process ($DATE_TIME) ---" | tee -a "$LOG_FILE"
echo "Log file: $LOG_FILE" | tee -a "$LOG_FILE"

# 1. Update Package Lists (Refresh list of available updates)
echo "Step 1/3: Updating package lists (apt update)..." | tee -a "$LOG_FILE"
if run_command "apt update -y"; then
    echo "Package lists refreshed successfully." | tee -a "$LOG_FILE"
else
    echo "FATAL ERROR: Failed to update package lists. Exiting." | tee -a "$LOG_FILE"
    exit 1
fi

# 2. Upgrade Installed Packages (Install updates)
echo "Step 2/3: Installing system upgrades (apt upgrade)..." | tee -a "$LOG_FILE"
# The '-y' flag assumes 'yes' to prompts, and '--with-new-pkgs' handles new dependencies.
if run_command "apt upgrade -y --with-new-pkgs"; then
    echo "System packages upgraded successfully." | tee -a "$LOG_FILE"
else
    echo "WARNING: Upgrade command encountered an issue, check log for details." | tee -a "$LOG_FILE"
fi

# 3. Clean up unused dependencies
echo "Step 3/3: Cleaning up orphaned dependencies (apt autoremove)..." | tee -a "$LOG_FILE"
if run_command "apt autoremove -y"; then
    echo "Orphaned dependencies removed successfully." | tee -a "$LOG_FILE"
else
    echo "Note: No dependencies to remove, or command failed." | tee -a "$LOG_FILE"
fi

# 4. Final Cleanup (Clear the package cache)
echo "Optional: Clearing local package cache (apt clean)..." | tee -a "$LOG_FILE"
run_command "apt clean"

echo "--- Update Process Finished ($DATE_TIME) ---" | tee -a "$LOG_FILE"

# Display a summary message
echo ""
echo "System update complete. Check $LOG_FILE for full details."
